```python
#!/usr/bin/env python3
"""
KiCAD Python: Load and inspect PCB
This script verifies your KiCAD Python setup works correctly
"""

import pcbnew
import os
import sys

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURATION - CHANGE THIS TO YOUR PROJECT PATH
# ═══════════════════════════════════════════════════════════════════════

PROJECT_DIR = "C:/Users/YourName/my_project/kicad_project"  # ← MODIFY THIS
# Or for Mac/Linux:
# PROJECT_DIR = "/home/username/my_project/kicad_project"

PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

# ═══════════════════════════════════════════════════════════════════════
# MAIN SCRIPT
# ═══════════════════════════════════════════════════════════════════════

def main():
    """Load and inspect PCB"""
    
    print("\n" + "="*70)
    print("KICAD PCB INSPECTOR")
    print("="*70)
    
    # Check if file exists
    if not os.path.exists(PCB_FILE):
        print(f"\n[✗] PCB file not found: {PCB_FILE}")
        print(f"\nMake sure:")
        print(f"  1. File exists at: {PCB_FILE}")
        print(f"  2. Update PROJECT_DIR in this script")
        sys.exit(1)
    
    # Load board
    print(f"\n[→] Loading: {PCB_FILE}")
    try:
        board = pcbnew.LoadBoard(PCB_FILE)
        print("[✓] PCB loaded successfully")
    except Exception as e:
        print(f"[✗] Failed to load: {e}")
        sys.exit(1)
    
    # Print board info
    print("\n" + "─"*70)
    print("BOARD PROPERTIES")
    print("─"*70)
    
    page_settings = board.GetPageSettings()
    width_mm = page_settings.GetWidthMM()
    height_mm = page_settings.GetHeightMM()
    
    print(f"Size: {width_mm:.1f}mm × {height_mm:.1f}mm")
    print(f"Copper layers: {board.GetCopperLayerCount()}")
    print(f"Total layers: {board.GetLayerCount()}")
    
    # Components
    footprints = board.GetFootprints()
    print(f"\nComponents: {len(footprints)}")
    print("\nFirst 10 components:")
    for i, fp in enumerate(footprints[:10]):
        print(f"  {i+1:2d}. {fp.GetReference():5s} - {fp.GetValue():20s} ({fp.GetFPID().GetLibItemName()})")
    
    if len(footprints) > 10:
        print(f"  ... and {len(footprints)-10} more")
    
    # Nets
    nets = board.GetNetsByName()
    print(f"\nNets: {len(nets)}")
    net_names = list(nets.keys())[:10]
    print(f"Sample nets: {', '.join(net_names)}")
    
    # Tracks
    tracks = board.GetTracks()
    vias = [t for t in tracks if isinstance(t, pcbnew.VIA)]
    traces = [t for t in tracks if not isinstance(t, pcbnew.VIA)]
    
    print(f"\nTraces: {len(traces)}")
    print(f"Vias: {len(vias)}")
    print(f"Total routing: {len(tracks)}")
    
    print("\n" + "="*70)
    print("[✓] Inspection complete!")
    print("="*70 + "\n")

if __name__ == "__main__":
    main()
```

#!/usr/bin/env python3
"""
KiCAD Python: Create new PCB project
Initializes blank board with design rules configured
"""

import pcbnew
import os

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURATION
# ═══════════════════════════════════════════════════════════════════════

PROJECT_DIR = "C:/Users/YourName/my_project/kicad_project"  # ← MODIFY THIS
PROJECT_NAME = "PhotonicAI"

# ═══════════════════════════════════════════════════════════════════════
# CREATE NEW BOARD
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Creating new PCB project...")

# Create empty board
board = pcbnew.BOARD()

# Set board size (100mm × 80mm)
page = board.GetPageSettings()
page.SetWidthMM(100)
page.SetHeightMM(80)
print("[✓] Board size: 100mm × 80mm")

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURE DESIGN RULES
# ═══════════════════════════════════════════════════════════════════════

design_rules = board.GetDesignSettings()

# Trace width minimum: 0.2mm
design_rules.m_TrackMinWidth = int(0.2e6)

# Clearance: 0.15mm
design_rules.m_MinClearance = int(0.15e6)

# Via size: 0.3mm drill, 0.6mm pad
design_rules.m_ViasMinSize = int(0.6e6)
design_rules.m_ViasMinDrill = int(0.3e6)
design_rules.m_ViasMinAnnulus = int(0.15e6)

print("[✓] Design rules configured:")
print("    • Min trace: 0.2mm")
print("    • Min clearance: 0.15mm")
print("    • Via drill: 0.3mm, pad: 0.6mm")

# ═════════════════════════════════════════════════════════════════════════
# CREATE POWER NETS
# ═════════════════════════════════════════════════════════════════════════

net_names = ["GND", "VCC_3.3V", "VCC_ANALOG", "VCC_DIGITAL"]

for net_name in net_names:
    net = pcbnew.NETINFO_ITEM(board, net_name)
    board.Add(net)

print(f"[✓] Created {len(net_names)} power nets")

# ═════════════════════════════════════════════════════════════════════════
# SAVE PROJECT
# ═════════════════════════════════════════════════════════════════════════

os.makedirs(PROJECT_DIR, exist_ok=True)
pcb_file = os.path.join(PROJECT_DIR, f"{PROJECT_NAME}.kicad_pcb")

board.Save(pcb_file)
print(f"\n[✓] Project created: {pcb_file}")
print(f"\nNext: Open in KiCAD to add components")

#!/usr/bin/env python3
"""
KiCAD Python: Automatically place components
Organizes 250+ components into functional regions
"""

import pcbnew
import os

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURATION
# ═══════════════════════════════════════════════════════════════════════

PROJECT_DIR = "C:/Users/YourName/my_project/kicad_project"  # ← MODIFY THIS
PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

# ═══════════════════════════════════════════════════════════════════════
# COMPONENT PLACEMENT DICTIONARY
# ═══════════════════════════════════════════════════════════════════════
# Reference: (X_mm, Y_mm, Rotation_degrees)

PLACEMENTS = {
    # TIA Op-Amps (8 channels, 4 dual op-amps)
    "U1": (30, 25, 0),
    "U2": (45, 25, 0),
    "U3": (30, 40, 0),
    "U4": (45, 40, 0),
    
    # Comparators (8 channels, 4 dual comparators)
    "U9": (70, 25, 0),
    "U10": (85, 25, 0),
    "U11": (70, 40, 0),
    "U12": (85, 40, 0),
    
    # DAC (central)
    "U13": (50, 15, 0),
    
    # Buffer op-amps
    "U14": (40, 15, 0),
    "U15": (60, 15, 0),
    
    # USB Controller
    "U18": (15, 40, 0),
    
    # Voltage regulator
    "U19": (50, 70, 0),
    
    # Passive components (auto-placed near their ICs)
    # Decoupling caps for each op-amp
    "C1": (32, 23, 0),
    "C2": (47, 23, 0),
    "C3": (32, 38, 0),
    "C4": (47, 38, 0),
    
    # Feedback resistors/caps (placed very close to op-amps)
    "R1": (31, 26, 90),
    "R2": (46, 26, 90),
    "R3": (31, 41, 90),
    "R4": (46, 41, 90),
}

# ═══════════════════════════════════════════════════════════════════════
# PLACEMENT FUNCTION
# ═══════════════════════════════════════════════════════════════════════

def place_component(board, reference, x_mm, y_mm, rotation_deg):
    """Place component at position"""
    
    # Find footprint
    footprint = board.FindFootprintByReference(reference)
    if footprint is None:
        print(f"[!] Not found: {reference}")
        return False
    
    # Convert mm to internal units
    x_units = int(x_mm * 1e6)
    y_units = int(y_mm * 1e6)
    
    # Set position
    footprint.SetPosition(pcbnew.VECTOR2I(x_units, y_units))
    
    # Set rotation (0, 900, 1800, 2700 for 0°, 90°, 180°, 270°)
    footprint.SetOrientation(
        pcbnew.EDA_ANGLE(rotation_deg * 10, pcbnew.DEGREES_T)
    )
    
    return True

# ═══════════════════════════════════════════════════════════════════════
# MAIN
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Loading board...")
board = pcbnew.LoadBoard(PCB_FILE)

print(f"[→] Placing {len(PLACEMENTS)} components...")

success_count = 0
for reference, (x, y, rotation) in PLACEMENTS.items():
    if place_component(board, reference, x, y, rotation):
        success_count += 1
        print(f"[✓] {reference:5s} at ({x:5.1f}, {y:5.1f})")
    else:
        print(f"[✗] {reference} - not found or error")

print(f"\n[✓] Placed {success_count}/{len(PLACEMENTS)} components")

# Save
board.Save(PCB_FILE)
print(f"[✓] Saved to: {PCB_FILE}")
print(f"\nNext: Open in KiCAD to verify placement")
```

#!/usr/bin/env python3
"""
KiCAD Python: Export manufacturing files
Generates Gerber, drill, BOM, and pick & place files
"""

import pcbnew
import os

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURATION
# ═══════════════════════════════════════════════════════════════════════

PROJECT_DIR = "C:/Users/YourName/my_project/kicad_project"  # ← MODIFY THIS
PROJECT_NAME = "PhotonicAI"
PCB_FILE = os.path.join(PROJECT_DIR, f"{PROJECT_NAME}.kicad_pcb")

OUTPUT_DIR = os.path.join(PROJECT_DIR, "../output")
GERBER_DIR = os.path.join(OUTPUT_DIR, "gerber")
DRILL_DIR = os.path.join(OUTPUT_DIR, "drill")
BOM_DIR = os.path.join(OUTPUT_DIR, "bom")

# Create directories
for d in [GERBER_DIR, DRILL_DIR, BOM_DIR]:
    os.makedirs(d, exist_ok=True)

# ═══════════════════════════════════════════════════════════════════════
# LOAD BOARD
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Loading board...")
board = pcbnew.LoadBoard(PCB_FILE)

# ═══════════════════════════════════════════════════════════════════════
# EXPORT GERBER FILES
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Exporting Gerber files...")

from pcbnew import PLOT_CONTROLLER

plotter = PLOT_CONTROLLER(board)
plot_options = plotter.GetPlotOptions()
plot_options.SetOutputDirectory(GERBER_DIR)
plot_options.SetFormat(pcbnew.PLOT_FORMAT_GERBER)

layers = [
    (pcbnew.F_Cu, "Front copper"),
    (pcbnew.B_Cu, "Back copper"),
    (pcbnew.F_Mask, "Front mask"),
    (pcbnew.B_Mask, "Back mask"),
    (pcbnew.F_SilkS, "Front silk"),
    (pcbnew.Edge_Cuts, "Board outline"),
]

for layer_id, description in layers:
    plotter.SetLayer(layer_id)
    plotter.PlotLayer()
    print(f"[✓] {description}")

plotter.ClosePlot()

# ═══════════════════════════════════════════════════════════════════════
# EXPORT DRILL FILE
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Exporting drill file...")

from pcbnew import EXCELLON_WRITER

drill_writer = EXCELLON_WRITER(board)
drill_writer.SetOptions(aMetricFmt=True, aMinimalHeader=False)
drill_writer.CreateDrillandMapFilesSet(DRILL_DIR, False, False)
print(f"[✓] Drill file exported")

# ═══════════════════════════════════════════════════════════════════════
# GENERATE BOM
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Generating BOM...")

bom_file = os.path.join(BOM_DIR, f"{PROJECT_NAME}_BOM.csv")
with open(bom_file, 'w') as f:
    f.write("Reference,Value,Footprint,X_mm,Y_mm\n")
    
    for fp in sorted(board.GetFootprints(), key=lambda x: x.GetReference()):
        f.write(f"{fp.GetReference()},")
        f.write(f"{fp.GetValue()},")
        f.write(f"{fp.GetFPID().GetLibItemName()},")
        f.write(f"{fp.GetX()/1e6:.2f},")
        f.write(f"{fp.GetY()/1e6:.2f}\n")

print(f"[✓] BOM exported: {bom_file}")

# ═══════════════════════════════════════════════════════════════════════
# GENERATE PICK & PLACE
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Generating Pick & Place file...")

pp_file = os.path.join(BOM_DIR, f"{PROJECT_NAME}_PickPlace.csv")
with open(pp_file, 'w') as f:
    f.write("Reference,Value,Footprint,X_mm,Y_mm,Rotation,Side\n")
    
    for fp in board.GetFootprints():
        side = "Back" if fp.IsFlipped() else "Front"
        f.write(f"{fp.GetReference()},")
        f.write(f"{fp.GetValue()},")
        f.write(f"{fp.GetFPID().GetLibItemName()},")
        f.write(f"{fp.GetX()/1e6:.2f},")
        f.write(f"{fp.GetY()/1e6:.2f},")
        f.write(f"{fp.GetOrientation().AsDegrees():.1f},")
        f.write(f"{side}\n")

print(f"[✓] Pick & Place: {pp_file}")

# ═══════════════════════════════════════════════════════════════════════
# SUMMARY
# ═══════════════════════════════════════════════════════════════════════

print("\n" + "="*70)
print("MANUFACTURING FILES READY!")
print("="*70)
print(f"\nOutput directory: {OUTPUT_DIR}")
print(f"\nFiles created:")
print(f"  ✓ Gerber files (6×): {GERBER_DIR}/")
print(f"  ✓ Drill file: {DRILL_DIR}/")
print(f"  ✓ BOM: {BOM_DIR}/{PROJECT_NAME}_BOM.csv")
print(f"  ✓ Pick & Place: {BOM_DIR}/{PROJECT_NAME}_PickPlace.csv")
print(f"\nNext: Zip and upload to PCBWay or JLCPCB")
print("="*70 + "\n")

#!/usr/bin/env python3
"""
KiCAD Python: Design Rule Check
Verifies PCB meets design specifications
"""

import pcbnew
import os

# ═══════════════════════════════════════════════════════════════════════
# CONFIGURATION
# ═══════════════════════════════════════════════════════════════════════

PROJECT_DIR = "C:/Users/YourName/my_project/kicad_project"  # ← MODIFY THIS
PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

# ═══════════════════════════════════════════════════════════════════════
# LOAD BOARD
# ═══════════════════════════════════════════════════════════════════════

print("\n[→] Loading board...")
board = pcbnew.LoadBoard(PCB_FILE)

# ═══════════════════════════════════════════════════════════════════════
# RUN CHECKS
# ═══════════════════════════════════════════════════════════════════════

print("[→] Running design checks...")

violations = []

# Check 1: Trace width
print("[→] Checking trace widths...")
for track in board.GetTracks():
    if not isinstance(track, pcbnew.PCB_VIA):
        width = track.GetWidth() / 1e6  # mm
        if width < 0.19:  # Less than 0.2mm
            violations.append(f"Trace too narrow: {width:.3f}mm")

# Check 2: Component overlap
print("[→] Checking component spacing...")
footprints = board.GetFootprints()
for i, fp1 in enumerate(footprints):
    for fp2 in footprints[i+1:]:
        x1, y1 = fp1.GetX(), fp1.GetY()
        x2, y2 = fp2.GetX(), fp2.GetY()
        distance = ((x2-x1)**2 + (y2-y1)**2) ** 0.5 / 1e6  # mm
        
        if distance < 3:  # Less than 3mm
            violations.append(
                f"Too close: {fp1.GetReference()} and {fp2.GetReference()} ({distance:.1f}mm)"
            )

# Check 3: Vias
print("[→] Checking vias...")
via_count = 0
for track in board.GetTracks():
    if isinstance(track, pcbnew.PCB_VIA):
        via_count += 1

# ═══════════════════════════════════════════════════════════════════════
# REPORT
# ═══════════════════════════════════════════════════════════════════════

print("\n" + "="*70)
if violations:
    print(f"⚠️  {len(violations)} VIOLATIONS FOUND")
    print("="*70)
    for v in violations[:10]:
        print(f"  • {v}")
    if len(violations) > 10:
        print(f"  ... and {len(violations)-10} more")
else:
    print("✅ ALL CHECKS PASSED!")
    print("="*70)

print(f"\nDesign summary:")
print(f"  Components: {len(board.GetFootprints())}")
print(f"  Traces: {len([t for t in board.GetTracks() if not isinstance(t, pcbnew.PCB_VIA)])}")
print(f"  Vias: {via_count}")
print(f"  Nets: {len(board.GetNetsByName())}")

print("="*70 + "\n")
```

═════════════════════════════════════════════════════════════════════════════════
PYTHON TIPS FOR KICAD
═════════════════════════════════════════════════════════════════════════════════

UNITS IN KICAD:
  • Internal unit: 1 unit ≈ 0.01 µm (very small)
  • 1 mm = 1,000,000 internal units
  • Always convert: value_mm * 1e6 → internal units
  
COORDINATES:
  • Origin (0,0) = top-left corner
  • X increases → right
  • Y increases → down

ANGLES:
  • Rotation in degrees × 10
  • 0° = 0
  • 90° = 900
  • 180° = 1800
  • 270° = 2700

LAYERS:
  pcbnew.F_Cu → Front copper
  pcbnew.B_Cu → Back copper
  pcbnew.In1_Cu → Inner layer 1
  pcbnew.In2_Cu → Inner layer 2
  pcbnew.F_Mask → Front solder mask
  pcbnew.Edge_Cuts → Board outline

USEFUL OBJECTS:
  board.GetFootprints() → List of all components

COMMON ERRORS & FIXES

ERROR: "ModuleNotFoundError: No module named 'pcbnew'"
FIX:
  Windows: Use "KiCAD Command Prompt" (installed with KiCAD)
  Or: C:\Program Files\KiCAD\bin\python.exe script.py
  Mac/Linux: Try: python3 script.py (usually works)

ERROR: "PCB file not found"
FIX:
  • Check PROJECT_DIR path in script
  • Verify file exists
  • Use absolute path, not relative

ERROR: "Component U1 not found"
FIX:
  • Component doesn't exist in PCB yet
  • In KiCAD Schematic: Schematic → Update PCB
  • Then run placement script

ERROR: "Cannot access property X"
FIX:
  • Object may be None
  • Add checks: if obj is not None: ...
  • Print object type for debugging

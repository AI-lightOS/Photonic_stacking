KICAD PYTHON API FUNDAMENTALS
═════════════════════════════════════════════════════════════════════════════════

The KiCAD API provides objects for everything in a PCB:

MAIN OBJECTS:
  • pcbnew.BOARD         → The PCB board itself
  • pcbnew.FOOTPRINT    → Component placement (contains pads)
  • pcbnew.TRACK        → Copper trace segment
  • pcbnew.VIA          → Via (hole connecting layers)
  • pcbnew.ZONE         → Copper fill area (planes)
  • pcbnew.PAD          → Connection point on footprint
  • pcbnew.TEVIA        → Test via
  
USEFUL PROPERTIES:
  • Position (X, Y coordinates)
  • Orientation (rotation angle)
  • Layer (which copper layer)
  • Width (trace thickness)
  • Net (electrical connection)
  • Keepout areas
  • Design rules (clearance, via size)

COMMON WORKFLOW:
  1. Load board from .kicad_pcb file
  2. Create footprints at specific positions
  3. Assign nets (electrical connectivity)
  4. Route traces between pads
  5. Add ground planes
  6. Run DRC (Design Rule Check)
  7. Export manufacturing files (Gerber, drill)

#!/usr/bin/env python3
"""
KiCAD Python Example - Load and inspect PCB
Usage: python3 00_hello_world.py
"""

import pcbnew
import os

# STEP 1: Define paths
PROJECT_DIR = "/path/to/my_photonic_project/kicad_project"
PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

# STEP 2: Load the PCB file
print("[→] Loading PCB file...")
board = pcbnew.LoadBoard(PCB_FILE)
print(f"[✓] PCB loaded: {PCB_FILE}")

# STEP 3: Inspect board properties
print("\n═══ BOARD INFORMATION ═══")
print(f"Board size: {board.GetPageSettings().GetWidthMils()} × {board.GetPageSettings().GetHeightMils()} mils")
print(f"Board layers: {board.GetLayerCount()}")
print(f"Copper layers: {board.GetCopperLayerCount()}")

# STEP 4: List all footprints (components)
print("\n═══ COMPONENTS (FOOTPRINTS) ═══")
footprints = board.GetFootprints()
print(f"Total footprints: {len(footprints)}")

for i, fp in enumerate(footprints[:5]):  # Show first 5
    print(f"  {i+1}. {fp.GetReference()}")
    print(f"     Footprint: {fp.GetFPID().GetLibItemName()}")
    print(f"     Position: {fp.GetX()}, {fp.GetY()} (in units)")
    print()

# STEP 5: List all nets (electrical connections)
print("\n═══ NETS (ELECTRICAL CONNECTIONS) ═══")
nets = board.GetNetsByName()
net_names = list(nets.keys())
print(f"Total nets: {len(net_names)}")
print(f"First 10 nets: {net_names[:10]}")

# STEP 6: List all traces
print("\n═══ TRACES (COPPER PATHS) ═══")
tracks = board.GetTracks()
print(f"Total tracks/traces: {len(tracks)}")

# Count by type
vias = [t for t in tracks if isinstance(t, pcbnew.VIA)]
traces = [t for t in tracks if not isinstance(t, pcbnew.VIA)]
print(f"  Traces: {len(traces)}")
print(f"  Vias: {len(vias)}")

print("\n[✓] Inspection complete!")

#!/usr/bin/env python3
"""
Create new KiCAD PCB project with design rules
"""

import pcbnew
import os

PROJECT_DIR = "/path/to/my_photonic_project/kicad_project"
PROJECT_NAME = "PhotonicAI"
OUTPUT_FILE = os.path.join(PROJECT_DIR, f"{PROJECT_NAME}.kicad_pcb")

# Create directory if needed
os.makedirs(PROJECT_DIR, exist_ok=True)

# ═════════════════════════════════════════════════════════════════════
# STEP 1: Create new board
# ═════════════════════════════════════════════════════════════════════
print("[→] Creating new board...")
board = pcbnew.BOARD()

# ═════════════════════════════════════════════════════════════════════
# STEP 2: Set board dimensions
# ═════════════════════════════════════════════════════════════════════
# 100mm × 80mm board (standard for our design)
# KiCAD uses internal units (1 unit ≈ 0.01 µm)

WIDTH_MM = 100
HEIGHT_MM = 80

# Convert mm to KiCAD units
# 1 mm = 1,000,000 internal units
WIDTH_UNITS = int(WIDTH_MM * 1e6)
HEIGHT_UNITS = int(HEIGHT_MM * 1e6)

page = board.GetPageSettings()
page.SetWidthMM(WIDTH_MM)
page.SetHeightMM(HEIGHT_MM)
print(f"[✓] Board size set: {WIDTH_MM}mm × {HEIGHT_MM}mm")

# ═════════════════════════════════════════════════════════════════════
# STEP 3: Set design rules (critical!)
# ═════════════════════════════════════════════════════════════════════
design_settings = board.GetDesignSettings()

# Trace width minimum: 0.2mm
MIN_TRACE = int(0.2e6)  # 0.2mm in units
design_settings.m_TrackMinWidth = MIN_TRACE

# Clearance (trace to trace): 0.15mm
MIN_CLEARANCE = int(0.15e6)
design_settings.m_MinClearance = MIN_CLEARANCE

# Via hole diameter: 0.3mm
# Via pad diameter: 0.6mm
VIA_DRILL = int(0.3e6)
VIA_PAD = int(0.6e6)
design_settings.m_ViasMinSize = VIA_PAD
design_settings.m_ViasMinDrill = VIA_DRILL

# Annular ring (pad around via): 0.15mm minimum
design_settings.m_ViasMinAnnulus = int(0.15e6)

print("[✓] Design rules configured")
print(f"    Min trace: 0.2mm")
print(f"    Min clearance: 0.15mm")
print(f"    Via drill: 0.3mm")
print(f"    Via pad: 0.6mm")

# ═════════════════════════════════════════════════════════════════════
# STEP 4: Configure layer stackup (4-layer PCB)
# ═════════════════════════════════════════════════════════════════════
# KiCAD default: F.Cu (front), GND, POWER, B.Cu (back)
# This is already set for 4 layers in our case

print(f"[✓] Configured {board.GetCopperLayerCount()} copper layers")

# ═════════════════════════════════════════════════════════════════════
# STEP 5: Set up nets (electrical connections)
# ═════════════════════════════════════════════════════════════════════
# Create basic power nets that will be used later
net_names = [
    "GND",
    "VCC_3.3V",
    "VCC_5V",
    "VCC_ANALOG",
    "AGND"
]

for net_name in net_names:
    # Get or create net
    net = board.FindNet(net_name)
    if net is None:
        # Create new net
        net = pcbnew.NETINFO_ITEM(board, net_name)
        board.Add(net)

print(f"[✓] Created {len(net_names)} power nets")

# ═════════════════════════════════════════════════════════════════════
# STEP 6: Save the board
# ═════════════════════════════════════════════════════════════════════
print(f"\n[→] Saving board to: {OUTPUT_FILE}")
board.Save(OUTPUT_FILE)
print(f"[✓] Project created successfully!")
print(f"\nYou can now open in KiCAD: {PROJECT_NAME}.kicad_pcb")
```

    footprint.SetOrientation(pcbnew.EDA_ANGLE(rotation * 10, pcbnew.DEGREES_T))

#!/usr/bin/env python3
"""
Route power distribution (VCC 3.3V and GND planes)
Creates copper areas (zones) for power planes
"""

import pcbnew
import os

PROJECT_DIR = "/path/to/my_photonic_project/kicad_project"
PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

print("[→] Loading board...")
board = pcbnew.LoadBoard(PCB_FILE)

# ═════════════════════════════════════════════════════════════════════
# FUNCTION: Create power plane (zone)
# ═════════════════════════════════════════════════════════════════════

def create_power_plane(board, net_name, layer, x1, y1, x2, y2):
    """
    Create copper fill area (plane) for power distribution
    
    Args:
        board: KiCAD board
        net_name: Net name (e.g., "VCC_3.3V" or "GND")
        layer: Layer ID (pcbnew.F_Cu, pcbnew.In1_Cu, etc.)
        x1, y1, x2, y2: Rectangle bounds in mm
    """
    
    # Convert mm to internal units
    x1_units = int(x1 * 1e6)
    y1_units = int(y1 * 1e6)
    x2_units = int(x2 * 1e6)
    y2_units = int(y2 * 1e6)
    
    # Find or create net
    net = board.FindNet(net_name)
    if net is None:
        net = pcbnew.NETINFO_ITEM(board, net_name)
        board.Add(net)
    
    # Create zone (copper fill area)
    zone = pcbnew.ZONE(board)
    zone.SetLayer(layer)
    zone.SetNet(net)
    zone.SetNetCode(net.GetNetCode())
    
    # Add rectangle outline
    outline = zone.Outline()
    outline.AddPoint(pcbnew.VECTOR2I(x1_units, y1_units))
    outline.AddPoint(pcbnew.VECTOR2I(x2_units, y1_units))
    outline.AddPoint(pcbnew.VECTOR2I(x2_units, y2_units))
    outline.AddPoint(pcbnew.VECTOR2I(x1_units, y2_units))
    
    # Set zone properties
    zone.SetLocalClearance(int(0.15e6))  # 0.15mm clearance
    zone.SetMinIslandArea(int((0.5e6) ** 2))  # 0.5mm² minimum island
    zone.SetPadConnection(pcbnew.ZONE_CONNECTION.THERMAL)  # Thermal relief pads
    
    # Add zone to board
    board.Add(zone)
    
    print(f"[✓] Created {net_name} plane on layer {layer}")
    print(f"    Rectangle: ({x1:.1f}, {y1:.1f}) to ({x2:.1f}, {y2:.1f})")

# ═════════════════════════════════════════════════════════════════════
# Create power planes on Layer 3 (POWER plane)
# ═════════════════════════════════════════════════════════════════════

print("[→] Creating power planes...")

# Layer 3 is "In2" in KiCAD (internal layer 2)
POWER_LAYER = pcbnew.In2_Cu

# GND plane (entire board bottom)
create_power_plane(board, "GND", POWER_LAYER, 0, 0, 100, 80)

# 3.3V_ANALOG plane (left half, analog section)
create_power_plane(board, "VCC_3.3V", POWER_LAYER, 0, 0, 50, 80)

# 3.3V_DIGITAL plane (right half, digital section)
create_power_plane(board, "VCC_DIGITAL", POWER_LAYER, 50, 0, 100, 80)

print("\n[→] Adding decoupling capacitors to GND via vias...")

# Create vias to connect decoupling caps to GND plane
# (In real design, these would be placed by component placement script)
# For now, just demonstrate via creation

def create_via(board, net_name, x_mm, y_mm, drill_size=0.3, pad_size=0.6):
    """
    Create a via at position (x, y)
    Connects from F.Cu to B.Cu through all layers
    """
    
    # Convert mm to internal units
    x_units = int(x_mm * 1e6)
    y_units = int(y_mm * 1e6)
    drill_units = int(drill_size * 1e6)
    
    # Find net
    net = board.FindNet(net_name)
    
    # Create via
    via = pcbnew.PCB_VIA(board)
    via.SetPosition(pcbnew.VECTOR2I(x_units, y_units))
    via.SetDrill(drill_units)
    via.SetWidth(int(pad_size * 1e6))
    via.SetNet(net)
    via.SetNetCode(net.GetNetCode())
    via.SetViaType(pcbnew.VIATYPE.THROUGH)
    
    # Add to board
    board.Add(via)
    
    return via

# Create ground vias under major IC locations
# TIA op-amp region
for x in [30, 45, 60, 75]:
    for y in [30, 45, 60]:
        create_via(board, "GND", x, y)

print("[✓] Added ground vias for GND plane connectivity")

# ═════════════════════════════════════════════════════════════════════
# Save updated board
# ═════════════════════════════════════════════════════════════════════

print("\n[→] Saving board with power planes...")

print("[✓] Power routing complete!")

print("  2. Select zones (power planes)")
print("  3. Right-click → Fill all zones")
print("  4. Verify no crossing nets")
```

#!/usr/bin/env python3
"""
Generate manufacturing files from KiCAD PCB
Exports: Gerber (copper layers), Drill file, BOM, Pick & Place
"""

import pcbnew
import os
from pathlib import Path

PROJECT_DIR = "/path/to/my_photonic_project/kicad_project"
PROJECT_NAME = "PhotonicAI"

#!/usr/bin/env python3
"""
Design Rule Check (DRC) - Verify PCB integrity
Checks for shorts, clearance violations, etc.
"""

import pcbnew
import os

PROJECT_DIR = "/path/to/my_photonic_project/kicad_project"
PCB_FILE = os.path.join(PROJECT_DIR, "PhotonicAI.kicad_pcb")

print("[→] Loading board...")
board = pcbnew.LoadBoard(PCB_FILE)

# ═════════════════════════════════════════════════════════════════════
# Run DRC (Design Rule Check)
# ═════════════════════════════════════════════════════════════════════

print("[→] Running Design Rule Check...")

from pcbnew import DRCRUNNER

# Create DRC runner
drc_runner = DRCRUNNER(board)

# Run checks
print("[→] Checking for violations...")

# Get design rules from board
design_rules = board.GetDesignSettings()

# Define checks to run
checks = [
    ("Copper-to-copper clearance", 0.15e-3),  # 0.15mm in meters
    ("Via to via spacing", 0.3e-3),           # 0.3mm
    ("Trace width minimum", 0.2e-3),          # 0.2mm
]

violations = []

# Check traces
print("\n[→] Checking traces...")
for track in board.GetTracks():
    # Get track properties
    width = track.GetWidth() / 1e6  # mm
    net = track.GetNet()
    
    # Check trace width
    if width < 0.2:
        violations.append(f"Trace too narrow: {width:.3f}mm (min 0.2mm)")

# Check footprints for overlap
print("[→] Checking component overlap...")
footprints = board.GetFootprints()
for i, fp1 in enumerate(footprints):
    for fp2 in footprints[i+1:]:
        # Check if footprints overlap (simplified)
        x1, y1 = fp1.GetX(), fp1.GetY()
        x2, y2 = fp2.GetX(), fp2.GetY()
        
        # Simple distance check
        distance = ((x2-x1)**2 + (y2-y1)**2) ** 0.5
        if distance < 5e6:  # 5mm minimum spacing
            violations.append(
                f"Components too close: {fp1.GetReference()} and {fp2.GetReference()}"
            )

# Report violations
print("\n" + "="*70)
if violations:
    print(f"⚠️  DRC VIOLATIONS FOUND: {len(violations)}")
    print("="*70)
    for i, violation in enumerate(violations, 1):
        print(f"{i}. {violation}")
else:
    print("✅ DRC PASSED - No violations found!")
    print("="*70)

print(f"\nDesign metrics:")
print(f"  • Total components: {len(board.GetFootprints())}")
print(f"  • Total traces: {len([t for t in board.GetTracks() if not isinstance(t, pcbnew.VIA)])}")
print(f"  • Total vias: {len([t for t in board.GetTracks() if isinstance(t, pcbnew.VIA)])}")
print(f"  • Total nets: {len(list(board.GetNetsByName().keys()))}")

if not violations:
    print("\n[✓] Design ready for fabrication!")
else:
    print("\n[✗] Please fix violations before sending to fab")
```

def route_signal(board, net_name, start_pad, end_pad, layer=pcbnew.F_Cu):
    """
    Create trace connecting two pads
    WARNING: Simple implementation (doesn't handle obstacles)
    For production: Use freerouting.org or professional routers
    """
    
    # Find pads
    start = None
    end = None
    
    for track in board.GetTracks():
        if hasattr(track, 'GetPad'):
            if track.GetPad() == start_pad:
                start = track
            if track.GetPad() == end_pad:
                end = track
    
    if start is None or end is None:
        return False
    
    # Create trace connecting them
    trace = pcbnew.PCB_TRACK(board)
    trace.SetStart(start.GetPosition())
    trace.SetEnd(end.GetPosition())
    trace.SetWidth(int(0.2e6))  # 0.2mm width
    trace.SetLayer(layer)
    
    # Find and set net
    net = board.FindNet(net_name)
    if net:
        trace.SetNet(net)
    
    board.Add(trace)
    return True
```

NOTE: Signal routing is complex (DFA algorithms)
RECOMMENDATION: Use KiCAD Freerouting or manual routing for critical signals
